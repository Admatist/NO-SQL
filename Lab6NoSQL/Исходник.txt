-- 1. Создание пространства ключей (Keyspace)

CREATE KEYSPACE IF NOT EXISTS lab6_shop

WITH replication = {'class': 'SimpleStrategy', 'replication_factor': 1};



USE lab6_shop;



-- 2. Создание таблиц (Query-driven design под Вариант 2)



-- Таблица 1: Заказы конкретного клиента (Q1, Q4, Q5)

CREATE TABLE IF NOT EXISTS orders_by_customer (

    customer_id text,

    order_date date,

    order_id text,

    status text,

    total_amount decimal,

    shipping_address text,

    PRIMARY KEY ((customer_id), order_date, order_id)

) WITH CLUSTERING ORDER BY (order_date DESC, order_id ASC);



-- Таблица 2: Заказы по статусу с Time Bucketing (Q2)

CREATE TABLE IF NOT EXISTS orders_by_status (

    status text,

    year_month text,

    order_date date,

    order_id text,

    customer_id text,

    total_amount decimal,

    PRIMARY KEY ((status, year_month), order_date, order_id)

) WITH CLUSTERING ORDER BY (order_date DESC, order_id ASC);



-- Таблица 3: Позиции внутри заказа (Q3)

CREATE TABLE IF NOT EXISTS order_items (

    order_id text,

    item_id text,

    product_name text,

    quantity int,

    price decimal,

    PRIMARY KEY ((order_id), item_id)

);
